public final class NuclearController extends TileEntity implements ITickable {
 public static final int CAPACITY=1000000, FE_PER_TICK=200, FUEL_TICKS=2400, WATER_CAPACITY=8000, SCRAM=1000, RESET=400, OUTPUT=400;
 public int energy,temperature,fuelTicks,production; public boolean enabled, scrammed, formed;
 private final AutoThermalControl autoThermal=new AutoThermalControl();
 public boolean autoTemperatureEnabled(){return autoThermal.enabled();}
 public boolean autoTemperatureInvalid(){return autoThermal.invalid();}
 public int autoTemperatureTarget(){return autoThermal.target();}
 // Server-side API only; no network handler/GUI is added by this experiment.
 public boolean setAutoTemperatureTarget(int target){
   if(world==null || world.isRemote || updating || reserved || filling || activeOperations>0 || removing)return false;
   if(!autoThermal.setTarget(target))return false;markDirty();return true;
 }
 public boolean setAutoTemperatureEnabled(boolean value){
   if(world==null || world.isRemote || updating || reserved || filling || activeOperations>0 || removing)return false;
   if(value && autoThermal.invalid())return false;
   autoThermal.setEnabled(value);markDirty();return true;
 }
 /** Diagnostic is instantaneous, not a guarantee of future stability. */
 public String autoTemperatureStatus(){
   if(autoThermal.invalid())return "INVALID_CONFIG";
   if(!autoThermal.enabled())return "OFF";
   if(scrammed || temperature>=SCRAM)return "PROTECTION";
   if(!enabled)return "REACTOR_OFF";
   if(!formed)return "STRUCTURE";
   if(coolantGuard && water()==0)return "NO_COOLANT_PROTECTION";
   if(!hasFuel())return "NO_FUEL";
   if(energy>CAPACITY-FE_PER_TICK)return "ENERGY_SATURATED";
   if(water()>0 && autoThermal.target()>0)return "COOLING_EXCEEDS_HEAT";
   return temperature==autoThermal.target()?"AT_TARGET":"ADJUSTING";
 }
 // Opt-in for old worlds: empty coolant trips a latched manual reset.
 private boolean coolantGuard;
 public boolean coolantGuard(){return coolantGuard;}
 void setCoolantGuard(boolean value){
   if(!usable())return;
   coolantGuard=value;
   if(value && enabled && water()==0)tripCoolantGuard();
   markDirty();
 }
 private void tripCoolantGuard(){
   enabled=false;scrammed=true;production=0;setIndicator(false);markDirty();
 }
