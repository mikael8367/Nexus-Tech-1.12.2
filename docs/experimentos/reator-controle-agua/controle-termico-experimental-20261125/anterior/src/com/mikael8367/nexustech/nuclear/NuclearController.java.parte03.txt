 fluid.amount<=0)return 0;
     // Bound before calling Forge's tank to avoid integer overflow on hostile offers.
     int amount=Math.min(fluid.amount,Math.max(0,WATER_CAPACITY-water()));
     if(amount==0)return 0;
     // Store plain water so tagged water cannot block normal pipes or buckets.
     FluidStack bounded=new FluidStack(FluidRegistry.WATER,amount);
     filling=true;activeOperations++;
     try{return tank.fill(bounded,doFill);}finally{
       filling=false;
       try{endOperation();}catch(RuntimeException ex){org.apache.logging.log4j.LogManager.getLogger("NexusTech").error("Nuclear removal finalization failed after fill; inspect recovery and inventory",ex);}
     }
   }
   public FluidStack drain(FluidStack fluid,boolean doDrain){return null;}
   public FluidStack drain(int amount,boolean doDrain){return null;}
 };

 private int extractEnergyChecked(int n,boolean sim,NuclearPart part,Object lease){
     if(part!=null && !part.validLease(this,lease))return 0;
     if(!usable() || reserved || n<=0 || validateStructure()!=1)return 0;
     if(part!=null && !part.validLease(this,lease))return 0;
     resetOutputForWorldTick();
     int out=Math.min(Math.min(n,energy),Math.max(0,OUTPUT-outputThisTick));
     if(!sim && out>0){
       bindStructure();
       if(!usable() || reserved || validateStructure()!=1)return 0;
       if(part!=null && !part.validLease(this,lease))return 0;
       resetOutputForWorldTick();out=Math.min(Math.min(n,energy),Math.max(0,OUTPUT-outputThisTick));
       energy-=out;outputThisTick+=out;markDirty();
     }return out;
 }
 private boolean usable(){
   if(removing || world==null || world.isRemote || isInvalid())return false;
   // Forge capabilities are synchronous: refuse calls off the real server thread
   // before reading world state. Lightweight test worlds are not WorldServer.
   if(world instanceof net.minecraft.world.WorldServer && (world.getMinecraftServer()==null || !world.getMinecraftServer().isCallingFromMinecraftThread()))return false;
   return world.isBlockLoaded(pos)&&world.getTileEntity(pos)==this;
 }

 /** Reclaim exact water from this position after replacing the controller. */
 void reclaimWater(){
   if(filling || bucketTransfer || activeOperations>0 || !usable())return;
   int room=WATER_CAPACITY-water();if(room<=0 || world.getPerWorldStorage()==null)return;
   NuclearRecovery recovery=NuclearRecovery.get(world);
   int amount=recovery.takeWater(pos,room);if(amount==0)return;
   // No external callbacks between debit and the plain tank assignment.
   tank.setFluid(new FluidStack(FluidRegistry.WATER,water()+amount));
   try{markDirty();}catch(RuntimeException ex){org.apache.logging.log4j.LogManager.getLogger("NexusTech").error("Nuclear reclaimed water notification failed",ex);}
 }
 /** Geometry uses controller as middle of front bottom row, front is its facing. */
 public static BlockPos at(BlockPos origin,EnumFacing front,int dx,int y,int depth){return origin.offset(front.rotateY(),dx).offset(front.getOpposite(),depth).up(y);}
 /** Synchronize the shared output budget to world time, not tile update order. */
 private void resetOutputForWorldTick(){
   long tick=world.getTotalWorldTime();
   if(outputTick!=tick){outputTick=tick;outputThisTick=0;}
 }
 /** Geometry only; deliberately ignores ownership so old owners can be checked without recursion. */
 private int validateGeometry(){
   if(!usable())return -1;
   IBlockState controller=world.getBlockState(pos);
   if(controller.getBlock()!=NuclearRegistry.CONTROLLER)return 0;
   EnumFacing front=controller.getValue(BlockHorizontal.FACING);
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++)
     if(!world.isBlockLoaded(at(pos,front,x,y,d)))return -1;
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++){
     BlockPos q=at(pos,front,x,y,d);
     if(x==0&&y==0&&d==0)continue;
     if(world.getBlockState(q).getBlock()!=(x==0&&y==2&&d==1?NuclearRegistry.CORE:NuclearRegistry.CASING)
         || !(world.getTileEntity(q) instanceof NuclearPart))return 0;
   }
   return 1;
 }
 private boolean containsPart(BlockPos part){
   IBlockState state=world.getBlockState(pos);
   if(state.getBlock()!=NuclearRegistry.CONTROLLER)return false;
   EnumFacing front=state.getValue(BlockHorizontal.FACING);
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++)
     if(!(x==0&&y==0&&d==0) && at(pos,front,x,y,d).equals(part))return true;
   return false;
 }
