     try {
       IEnergyStorage receiver=other.getCapability(CapabilityEnergy.ENERGY,side.getOpposite());
       if(receiver==null || !receiver.canReceive())continue;

       if(!usable() || validateStructure()!=1)return;

       resetOutputForWorldTick();offer=Math.min(energy,OUTPUT-outputThisTick);if(offer<=0)return;
       energy-=offer;outputThisTick+=offer;reserved=true;
       int got=clamp(receiver.receiveEnergy(offer,false),offer);
       energy+=offer-got;outputThisTick-=offer-got;
     } catch(RuntimeException ex) {
       // A receiver may have accepted the reservation before throwing. Never refund it.
     } finally {reserved=false;}
   }

 private void readInventory(NBTTagCompound n){
   recoveryOverflow.clear();
   NBTTagList savedOverflow=n.getTagList("RecoveryOverflow",10);
   for(int i=0;i<savedOverflow.tagCount();i++){
     ItemStack stack=new ItemStack(savedOverflow.getCompoundTagAt(i));
     if(!stack.isEmpty())recoveryOverflow.add(stack);
   }
   NBTTagCompound source=n.getCompoundTag("Inventory"),fixed=new NBTTagCompound();
   NBTTagList items=source.getTagList("Items",10),normal=new NBTTagList();
   boolean[] occupied=new boolean[3];
   for(int i=0;i<items.tagCount();i++){
     NBTTagCompound tag=items.getCompoundTagAt(i);int slot=tag.getInteger("Slot");
     ItemStack stack=new ItemStack(tag);
     if(stack.isEmpty())continue;
     if(slot>=0 && slot<3 && !occupied[slot]){occupied[slot]=true;normal.appendTag(tag.copy());}
     else recoveryOverflow.add(stack);
   }
   fixed.setInteger("Size",3);fixed.setTag("Items",normal);inventory.deserializeNBT(fixed);
 }
