   public ItemStack extractItem(int slot,int amount,boolean simulate){
     if(bucketTransfer || automationExtracting || activeOperations>0 || !usable())return ItemStack.EMPTY;
     if(slot!=2){inventory.getStackInSlot(slot);return ItemStack.EMPTY;}
     if(amount<=0)return ItemStack.EMPTY;
     // Copies may invoke foreign capabilities: defer removal, block nested operations.
     automationExtracting=true;activeOperations++;
     try{
       ItemStack original=inventory.getStackInSlot(slot);
       if(original.isEmpty())return ItemStack.EMPTY;
       int count=Math.min(amount,Math.min(original.getCount(),original.getMaxStackSize()));
       if(count<=0 || !usable())return ItemStack.EMPTY;
       ItemStack result=original.copy();result.setCount(count);
       if(!usable())return ItemStack.EMPTY;
       if(simulate)return result;
       ItemStack remaining=ItemStack.EMPTY;
       if(count<original.getCount()){
         remaining=original.copy();remaining.shrink(count);
         if(!usable())return ItemStack.EMPTY;
       }
       // Commit before notifying: do not undo a successful transfer on callback failure.
       recoveryInventory.setWithoutNotification(slot,remaining);
       try{markDirty();}catch(RuntimeException ex){org.apache.logging.log4j.LogManager.getLogger("NexusTech").error("Nuclear extraction notification failed after mutation; chunk may not be marked dirty",ex);}
       return result;
     }finally{
       try{endOperation();}catch(RuntimeException ex){org.apache.logging.log4j.LogManager.getLogger("NexusTech").error("Nuclear removal finalization failed after extraction; inspect recovery and inventory",ex);}
       finally{automationExtracting=false;}
     }
   }
   public int getSlotLimit(int slot){return inventory.getSlotLimit(slot);}
