 /** Pure check: -1 unloaded, 0 invalid or owned, 1 claimable. No bind or dirtying. */
 public int validateStructure(){
   int geometry=validateGeometry(); if(geometry!=1)return geometry;
   EnumFacing front=world.getBlockState(pos).getValue(BlockHorizontal.FACING);
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++){
     if(x==0&&y==0&&d==0)continue;
     BlockPos q=at(pos,front,x,y,d);
     BlockPos owner=((NuclearPart)world.getTileEntity(q)).owner;
     if(owner!=null&&!owner.equals(pos)){
       if(!world.isBlockLoaded(owner))return -1;
       TileEntity old=world.getTileEntity(owner);
       if(old instanceof NuclearController){
         NuclearController c=(NuclearController)old;
         // Damaged but hot/armed owners keep their bindings until safe.
         if(c.enabled || c.temperature>RESET)return 0;
         int oldGeometry=c.validateGeometry();
         if(oldGeometry<0)return -1;
         if(oldGeometry==1 && c.containsPart(q))return 0;
       }
     }
   }
   return 1;
 }
 /** Called only after a successful pure validation, on an actual server operation. */
 private void bindStructure(){
   if(validateStructure()!=1)return;
   EnumFacing front=world.getBlockState(pos).getValue(BlockHorizontal.FACING);
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++){
     if(x==0&&y==0&&d==0)continue;
     BlockPos q=at(pos,front,x,y,d);
     if(!usable() || !world.isBlockLoaded(q))return;
     TileEntity t=world.getTileEntity(q);if(!(t instanceof NuclearPart))return;
     ((NuclearPart)t).bind(pos);
   }
 }
 /** Claim and publish a valid structure before opening from any of its parts. */
 boolean prepareForGui(){
   if(validateStructure()!=1)return false;
   bindStructure();
   if(!allPartsBound()){stopForStructureLoss();return false;}
   formed=true;
   setVisualFormed(true);
   syncPartVisuals(true);
   if(!allPartsBound()){stopForStructureLoss();return false;}
   return true;
 }
 private boolean allPartsBound(){
   if(validateStructure()!=1)return false;
   EnumFacing front=world.getBlockState(pos).getValue(BlockHorizontal.FACING);
   for(int x=-1;x<=1;x++)for(int y=0;y<5;y++)for(int d=0;d<3;d++){
     if(x==0&&y==0&&d==0)continue;
     BlockPos q=at(pos,front,x,y,d);
     if(!world.isBlockLoaded(q))return false;
     TileEntity t=world.getTileEntity(q);
     if(!(t instanceof NuclearPart) || t.isInvalid() || !pos.equals(((NuclearPart)t).owner))return false;
   }
   return true;
 }

 private void acceptBucket(){
   ItemStack in=inventory.getStackInSlot(1),out=inventory.getStackInSlot(2);
   ItemStack empty=new ItemStack(Items.BUCKET);
   if(in.isEmpty() || in.getItem()!=Items.WATER_BUCKET || tank.getCapacity()-water()<1000 || (!out.isEmpty() && (!ItemStack.areItemsEqual(out,empty) || !ItemStack.areItemStackTagsEqual(out,empty) || out.getCount()>=out.getMaxStackSize())))return;
   // Tick operation defers removal. Copies may consult modded item capabilities,
   // so lock before preparing them and abandon the transfer if removal intervenes.
   bucketTransfer=true;
   try {
     ItemStack nextIn=in.copy();nextIn.shrink(1);if(nextIn.isEmpty())nextIn=ItemStack.EMPTY;
     ItemStack nextOut=out.isEmpty()?empty:out.copy();if(!out.isEmpty())nextOut.grow(1);
     FluidStack nextWater=new FluidStack(FluidRegistry.WATER,water()+1000);
     if(removing || !usable())return;

     tank.setFluid(nextWater);
     recoveryInventory.setWithoutNotification(1,nextIn);
     recoveryInventory.setWithoutNotification(2,nextOut);
     // Mutation is committed even if a synchronous world callback throws.
     try{markDirty();}catch(RuntimeException ex){org.apache.logging.log4j.LogManager.getLogger("NexusTech").error("Nuclear bucket conversion notification failed after mutation; chunk may not be marked dirty",ex);}
   } finally {bucketTransfer=false;}
 }
